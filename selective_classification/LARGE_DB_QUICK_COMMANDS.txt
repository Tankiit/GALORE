    --max_samples 30000 \

    --device cuda

    --max_samples 30000 \
    --use_faiss \
    --device cuda

    --data_dir ./datacomp_data \

    --device cuda

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Fast (GPU, large batch):
  --device cuda --batch_size 512

Balanced (GPU, medium batch):
  --device cuda --batch_size 256

Memory-safe (GPU, small batch):
  --device cuda --batch_size 64

CPU (slow but works anywhere):
  --device cpu --batch_size 32

Large memory (bigger chunks):
  --chunk_size 20000

Small memory (smaller chunks):
  --chunk_size 5000

TROUBLESHOOTING
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Out of memory?
  → Reduce batch_size: --batch_size 32
  → Reduce chunk_size: --chunk_size 5000
  → Use CPU: --device cpu

Too slow?
  → Use GPU: --device cuda
  → Increase batch: --batch_size 512
  → Fewer samples: --max_samples 10000

FAISS not installing?
  → conda install -c pytorch faiss-gpu
  → Or pip install faiss-cpu
  → Or disable: (remove --use_faiss flag)

Build interrupted?
  → Resume: python build_large_retrieval_database.py --output_dir ... --resume
